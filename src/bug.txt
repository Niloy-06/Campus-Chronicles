#include <stdio.h>

void printNumbers(int numbers[], int size)
{
    for (int i = 0; i < size; i++)
    {
        printf("%d ", numbers[i]);
    }
}

int main()
{
    int numbers[] = {10, 20, 30, 40, 50};
    int size = 5;

    printNumbers(numbers, size);

    int *ptr = &numbers;

    printf("\nFirst number: %d\n", *ptr);
    printf("Second number: %d\n", *(ptr + 1));

    return 0;
}
